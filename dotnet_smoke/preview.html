#[diplomat::bridge]
pub mod ffi {
    #[diplomat::opaque_mut]
    pub struct Color {
        brightness: u8,
    }

    impl Color {
        pub fn new() -> Box<Color> {
            Box::new(Color { brightness: 0 })
        }

        pub fn brightness(&self) -> u8 {
            self.brightness
        }

        pub fn set_brightness(&mut self, value: u8) {
            self.brightness = value;
        }
    }
}
